1"""Application settings, loaded from environment / .env. 

2 

3qmd-py lives in its own `qmd_py` Postgres schema inside the same `qmd` 

4database the TS/Postgres reference implementation uses (not a separate 

5database - see the project plan's "Single Postgres, forever" decision). 

6Isolation from the TS tables (which live in `public`) comes from the 

7`search_path` connection option below, not from a different database. 

8""" 

9 

10from functools import lru_cache 

11from urllib.parse import quote 

12 

13from pydantic_settings import BaseSettings, SettingsConfigDict

21 it's appended automatically in `sqlalchemy_url`."""

38 @property 

39 def sqlalchemy_url(self) -> str: 

40 # Deliberately just our own schema, NOT `qmd_py,public`. TSVECTOR is 

41 # a built-in Postgres type (doesn't need `public` on the path at 

42 # all), and a multi-entry search_path breaks Alembic autogenerate in 

43 # a subtle way: Postgres's `pg_table_is_visible()` (which 

44 # SQLAlchemy's dialect uses for "reflect the default/implicit 

45 # schema") considers a table visible if ANY search_path entry 

46 # resolves to it - so with `qmd_py,public` on the path, the TS 

47 # reference implementation's tables in `public` got swept into 

48 # autogenerate's comparison and it proposed dropping them, even 

49 # though `env.py`'s `include_name` schema-level filter correctly 

50 # excluded `public` from the *explicit* multi-schema reflection 

51 # pass. That's a separate pass from the "default schema" one, and 

52 # only the multi-entry search_path triggers it. 

53 # Once Phase 5 needs pgvector's `vector` type (extension-provided, 

54 # unlike TSVECTOR), revisit: either re-add `public` here and solve 

55 # the autogenerate issue some other way (e.g. schema-qualifying the 

56 # vector type explicitly), or install `pgvector` into `qmd_py` 

57 # itself so this schema never needs `public` on its search_path. 

58 options = quote(f"-c search_path={self.postgres_schema}") 

59 separator = "&" if "?" in self.postgres_url else "?" 

60 return f"{self.postgres_url}{separator}options={options}" 

61 

62 

63@lru_cache 

64def get_settings() -> Settings: 

65 """Lazily constructed so importing this module doesn't require 

66 MARQ_POSTGRES_URL to already be set (e.g. during test collection).""" 

67 return Settings() # type: ignore[call-arg] # postgres_url comes from env
