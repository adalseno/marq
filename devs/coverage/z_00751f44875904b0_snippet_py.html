1"""Snippet extraction for search-result output - port of the TS 

2reference's `extractSnippet`/`extractIntentTerms` (src/store.ts). 

3""" 

4 

5import re 

6from dataclasses import dataclass 

7 

8_INTENT_STOP_WORDS = frozenset( 

9 { 

10 # 2-char function words 

11 "am", "an", "as", "at", "be", "by", "do", "he", "if", 

12 "in", "is", "it", "me", "my", "no", "of", "on", "or", "so", 

13 "to", "up", "us", "we", 

14 # 3-char function words 

15 "all", "and", "any", "are", "but", "can", "did", "for", "get", 

16 "has", "her", "him", "his", "how", "its", "let", "may", "not", 

17 "our", "out", "the", "too", "was", "who", "why", "you", 

18 # 4+ char common words 

19 "also", "does", "find", "from", "have", "into", "more", "need", 

20 "show", "some", "tell", "that", "them", "this", "want", "what", 

21 "when", "will", "with", "your", 

22 # Search-context noise 

23 "about", "looking", "notes", "search", "where", "which", 

24 } 

25) # fmt: skip 

26 

27_PUNCT_STRIP = re.compile(r"^[^\w]+|[^\w]+$", re.UNICODE) 

28 

29 

30def extract_intent_terms(intent: str) -> list[str]: 

31 terms = [] 

32 for token in intent.lower().split(): 

33 stripped = _PUNCT_STRIP.sub("", token) 

34 if len(stripped) > 1 and stripped not in _INTENT_STOP_WORDS: 

35 terms.append(stripped) 

36 return terms 

37 

38 

39INTENT_WEIGHT_SNIPPET = 0.3 

40 

41# Mirrors the TS reference's char-based chunk-size constant used as the 

42# fallback search window when a chunk's own length isn't known. 

43_CHUNK_SIZE_CHARS = 900 * 3 

44 

45 

46@dataclass 

47class SnippetResult: 

48 line: int 

49 snippet: str 

50 lines_before: int 

51 lines_after: int 

52 snippet_lines: int 

53 

54 

55def extract_snippet( 

56 body: str, 

57 query: str, 

58 max_len: int = 500, 

59 chunk_pos: int | None = None, 

60 chunk_len: int | None = None, 

61 intent: str | None = None, 

62) -> SnippetResult: 

63 total_lines = body.count("\n") + 1 

64 search_body = body 

65 line_offset = 0 

66 

67 if chunk_pos is not None and chunk_pos >= 0: 

68 search_len = chunk_len or _CHUNK_SIZE_CHARS 

69 context_start = max(0, chunk_pos - 100) 

70 context_end = min(len(body), chunk_pos + search_len + 100) 

71 search_body = body[context_start:context_end] 

72 if context_start > 0: 

73 line_offset = body[:context_start].count("\n") 

74 

75 lines = search_body.split("\n") 

76 query_terms = [t for t in query.lower().split() if t] 

77 intent_terms = extract_intent_terms(intent) if intent else [] 

78 best_line = 0 

79 best_score = -1.0 

80 

81 for i, line in enumerate(lines): 

82 line_lower = line.lower() 

83 score = 0.0 

84 for term in query_terms: 

85 if term in line_lower: 

86 score += 1.0 

87 for term in intent_terms: 

88 if term in line_lower: 

89 score += INTENT_WEIGHT_SNIPPET 

90 if score > best_score: 

91 best_score = score 

92 best_line = i 

93 

94 if chunk_pos is not None and chunk_pos >= 0 and best_score <= 0: 

95 if chunk_pos == 0: 

96 return extract_snippet(body, query, max_len, None, None, intent) 

97 context_start = max(0, chunk_pos - 100) 

98 best_line = ( 

99 search_body[: chunk_pos - context_start].count("\n") if chunk_pos > context_start else 0 

100 ) 

101 

102 start = max(0, best_line - 1) 

103 end = min(len(lines), best_line + 3) 

104 snippet_lines_list = lines[start:end] 

105 snippet_text = "\n".join(snippet_lines_list) 

106 

107 if chunk_pos and chunk_pos > 0 and not snippet_text.strip(): 

108 return extract_snippet(body, query, max_len, None, None, intent) 

109 

110 if len(snippet_text) > max_len: 

111 snippet_text = snippet_text[: max_len - 3] + "..." 

112 

113 absolute_start = line_offset + start + 1 # 1-indexed 

114 snippet_line_count = len(snippet_lines_list) 

115 lines_before = absolute_start - 1 

116 lines_after = total_lines - (absolute_start + snippet_line_count - 1) 

117 

118 header = ( 

119 f"@@ -{absolute_start},{snippet_line_count} @@ " 

120 f"({lines_before} before, {lines_after} after)" 

121 ) 

122 snippet = f"{header}\n{snippet_text}" 

123 

124 return SnippetResult( 

125 line=line_offset + best_line + 1, 

126 snippet=snippet, 

127 lines_before=lines_before, 

128 lines_after=lines_after, 

129 snippet_lines=snippet_line_count, 

130 )
